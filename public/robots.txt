# Goal: keep this CV out of search results and out of bulk/AI scrapes.
#
# The HTML pages carry <meta name="robots" content="noindex, nofollow">, and
# that is what actually removes them from search results. A crawler has to
# fetch a page to read that tag, so search engines are deliberately NOT
# blocked here — "Disallow: /" would hide the noindex and let a bare URL be
# listed anyway. The PDFs cannot carry a meta tag, so they are blocked instead.

User-agent: *
Disallow: /*.pdf$
